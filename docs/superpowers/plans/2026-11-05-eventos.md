# Eventos rotativos (fase 2) — Plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.
>
> **NOTA del proyecto (AGENTS.md):** la usuaria hace los commits. Ninguna tarea ejecuta `git`.

**Goal:** Eventos rotativos de 1–4 días (rotación determinista sin cron) con tareas diarias recompensadas en tinta y tienda de coleccionables (pegatinas y cartas de evento) que la gasta.

**Architecture:** `EventService` calcula el evento activo por fecha (época fija + ciclo), registra progreso (`event_progress`) y vende coleccionables (`user_collectibles`) en transacción. Hooks silenciosos en las acciones existentes (balota, lectura, libro, álbum). Página nueva `Events/Index` con entrada propia en el menú.

**Tech Stack:** Laravel 13, Inertia + Vue 3, Tailwind 4, Pest, Vitest.

**Spec:** `docs/superpowers/specs/2026-11-05-eventos-design.md`

- [ ] **Step 4: Crear `app/Services/EventService.php` (solo rotación)**

```php
<?php

namespace App\Services;

use Carbon\CarbonImmutable;

class EventService
{
    public function today(): CarbonImmutable
    {
        return CarbonImmutable::now(config('events.timezone'))->startOfDay();
    }

    public function activeEvent(?CarbonImmutable $day = null): array
    {
        $day ??= $this->today();
        $catalog = config('events.catalog');
        $epoch = CarbonImmutable::parse(config('events.epoch'), config('events.timezone'))->startOfDay();

        $dayNumber = $epoch->diffInDays($day);
        $cycleDays = array_sum(array_column($catalog, 'days'));
        $offset = $dayNumber % $cycleDays;

        foreach ($catalog as $event) {
            if ($offset < $event['days']) {
                return [
                    'event' => $event,
                    'day' => $offset + 1,
                    'days' => $event['days'],
                ];
            }

            $offset -= $event['days'];
        }

        return [
            'event' => $catalog[0],
            'day' => 1,
            'days' => $catalog[0]['days'],
        ];
    }
}
```

Nota: Carbon 3 `diffInDays` puede devolver float negativo si `$day < $epoch`; en producción nunca ocurre (época en el pasado) y en pruebas se usan fechas posteriores.

```php
use App\Models\EventProgress;
use App\Models\User;
use App\Models\UserWallet;

public function todayProgress(User $user): array
{
    $active = $this->activeEvent();

    return EventProgress::query()
        ->where('user_id', $user->id)
        ->where('event_key', $active['event']['key'])
        ->whereDate('day_date', $this->today()->toDateString())
        ->get()
        ->all();
}

public function completeTask(User $user, string $taskKey): void
{
    try {
        $active = $this->activeEvent();
        $event = $active['event'];

        if (! in_array($taskKey, $event['tasks'], true)) {
            return;
        }

        $reward = (int) config("events.tasks.{$taskKey}.ink", 0);

        DB::transaction(function () use ($user, $event, $reward, $taskKey) {
            EventProgress::firstOrCreate([
                'user_id' => $user->id,
                'event_key' => $event['key'],
                'day_date' => $this->today()->toDateString(),
                'task_key' => $taskKey,
            ], ['ink_earned' => $reward]);

            if ($reward > 0) {
                $wallet = UserWallet::firstOrCreate(['user_id' => $user->id]);
                $wallet->increment('ink', $reward);
            }
        });
    } catch (Throwable) {
        return;
    }
}
```

```php
use App\Models\UserCollectible;

public function shop(): array
{
    return $this->activeEvent()['event'];
}

public function collection(User $user): array
{
    return $user->collectibles()->orderByDesc('acquired_at')->get()->all();
}

public function findItem(string $key): ?array
{
    $event = $this->activeEvent()['event'];

    foreach ($event['stickers'] as $sticker) {
        if ($sticker['key'] === $key) {
            return $sticker + ['kind' => 'sticker'];
        }
    }

    foreach ($event['cards'] as $card) {
        if ($card['key'] === $key) {
            return $card + ['kind' => 'card'];
        }
    }

    return null;
}

public function grantCollectible(User $user, string $key, string $source): UserCollectible
{
    $item = $this->findItem($key);

    return UserCollectible::create([
        'user_id' => $user->id,
        'collectible_key' => $key,
        'kind' => $item['kind'],
        'event_key' => $this->activeEvent()['event']['key'],
        'source' => $source,
        'acquired_at' => now(),
    ]);
}

public function buyCollectible(User $user, string $key): array
{
    $item = $this->findItem($key);

    if ($item === null) {
        throw ValidationException::withMessages(['item' => __('ui.events_item_missing')]);
    }

    $owned = UserCollectible::query()
        ->where('user_id', $user->id)
        ->where('collectible_key', $key)
        ->exists();

    if ($owned) {
        abort(409);
    }

    $wallet = UserWallet::firstOrCreate(['user_id' => $user->id]);

    if ($wallet->ink < $item['cost']) {
        throw ValidationException::withMessages(['item' => __('ui.events_not_enough_ink')]);
    }

    return DB::transaction(function () use ($user, $item, $wallet) {
        $wallet->decrement('ink', $item['cost']);
        $collectible = $this->grantCollectible($user, $item['key'], 'shop');

        return [
            'collectible' => ['key' => $collectible->collectible_key, 'kind' => $collectible->kind],
            'ink' => (int) $wallet->fresh()->ink,
        ];
    });
}
```

```php
        $tasks = collect($event['tasks'])->map(function (string $taskKey) use ($events, $user) {
            $done = collect($events->todayProgress($user))->contains('task_key', $taskKey);

            return [
                'key' => $taskKey,
                'ink' => (int) config("events.tasks.{$taskKey}.ink", 0),
                'done' => $done,

        })->values();

        $shop = collect([...$event['stickers'], ...$event['cards']])
            ->map(fn (array $item) => $item + ['owned' => in_array($item['key'], $owned, true)])
            ->values();

        return Inertia::render('Events/Index', [
            'event' => [
                'key' => $event['key'],
                'name' => $event['name'],
                'theme' => $event['theme'],
                'day' => $active['day'],
                'days' => $active['days'],
            ],
            'tasks' => $tasks,
            'shop' => $shop,
            'collection' => collect($events->collection($user))->map(fn ($c) => [
                'key' => $c->collectible_key,
                'kind' => $c->kind,
                'eventKey' => $c->event_key,
            ])->values(),
            'ink' => (int) ($user->wallet?->ink ?? 0),
        ]);
    }

    public function buy(Request $request, EventService $events): JsonResponse
    {
        $data = $request->validate(['item' => ['required', 'string', 'max:64']]);

        return response()->json($events->buyCollectible($request->user(), $data['item']));
    }
}
```

```vue
async function buy(itemKey) {
    error.value = '';

    try {
        const { data } = await axios.post('/events/buy', { item: itemKey });
        ink.value = data.ink;

        if (! collection.value.some((c) => c.key === data.collectible.key)) {

        }
    } catch {

    }
}
```
